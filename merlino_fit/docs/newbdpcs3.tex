}

\section{Updated BDPCS3 formulation}
BDPCS3 is a compact, localized \emph{a~posteriori} correction applied to DPCS3 equilibrium bond lengths. It is designed to (i) mimic the missing core--valence (CV) correlation and (ii) mitigate the mild overestimation of valence conjugation typical of semi-local functionals.
In the original formulation, bonded pairs were selected via a hard cutoff, yielding a discontinuous correction. Here the bond detection is regularized by the smooth $\erf$-based cutoff, and all correction terms are continuous functions of $r$ that decay asymptotically to zero, enabling a well-behaved dissociation limit.

We define the single-atom indicators
\[
\chi_{\mathrm{CS}}(i)=\dZi{6}+0.5\,\dZi{16},\qquad
\chi_{\mathrm{CS}}(j)=\dZj{6}+0.5\,\dZj{16},
\]
\[
\chi_{\mathrm{CH}}(i)=\dZi{6}+\dZi{1},\qquad
\chi_{\mathrm{CH}}(j)=\dZj{6}+\dZj{1},
\]
with \(\delta_{a,b}\) the Kronecker delta. The updated correction reads
\begin{equation}
  r_{ij}^{\mathrm{BDPCS3}} 
  = r_{ij}^{\mathrm{DPCS3}} 
  + A_{n_i,n_j}\!
    \left[
      1 - \chi_{\mathrm{CS}}(i)\,\chi_{\mathrm{CS}}(j)\,
      \exp\!\left(
        -\left(\frac{ r_{ij}^{\mathrm{DPCS3}} - r_{Z_i,Z_j}^{\mathrm{cov}} + \Delta r }{ \sigma }\right)^{\!2}
      \right)
    \right],
\end{equation}
with
\begin{equation}
  A_{n_i,n_j} 
  = -K\,r_{Z_i,Z_j}^{\mathrm{cov}}
    \left\{1 + 0.25\,\chi_{\mathrm{CH},ij}\right\}
    \,[\max(n_i,n_j)^2-1]\,
    f_{\mathrm{coord}}\!\bigl(r_{ij}^{\mathrm{DPCS3}}\bigr),
\end{equation}
and the long-range cutoff
\begin{equation}
  f_{\mathrm{coord}}(r) =
  \frac{1}{2}\left[\,1 - \erf\!\left(\frac{r-(r_{Z_i,Z_j}^{\mathrm{cov}}+0.35)}{\sigma}\right)\right],
\end{equation}
so that \(\Delta r(r)\!\to\!0\) for \(r\gtrsim r_{Z_i,Z_j}^{\mathrm{cov}}+0.35\)~\AA.
\begin{align}
  \chi_{\mathrm{CS}}(x)&=\delta_{Z_x,6}+0.5\,\delta_{Z_x,16},\\
  \chi_{\mathrm{CH},ij}&=\delta_{Z_i,6}\delta_{Z_j,1}+\delta_{Z_i,1}\delta_{Z_j,6}.
\end{align}

Optimized parameters: \(K=0.006\), \(\Delta r=0.17\)~\AA, \(\sigma=0.057\)~\AA.  
Covalent radii (Pyykk\"o--Atsumi, \AA): H 0.31, C 0.76, O 0.66, S 1.05.

\begin{tikzpicture}
  % Parameters
  \pgfmathsetmacro{\K}{0.006}
  \pgfmathsetmacro{\dr}{0.17}
  \pgfmathsetmacro{\sig}{0.057}

  % Covalent sums
  \pgfmathsetmacro{\rcovCH}{1.07}
  \pgfmathsetmacro{\rcovCC}{1.52}
  \pgfmathsetmacro{\rcovCO}{1.42}
  \pgfmathsetmacro{\rcovCS}{1.81}

  % Prefactors {1+0.25*chi_CH} and [max(n)^2-1]:
  \pgfmathsetmacro{\prefCH}{1.25} \pgfmathsetmacro{\nfacCH}{3}
  \pgfmathsetmacro{\prefCC}{1.00} \pgfmathsetmacro{\nfacCC}{3}
  \pgfmathsetmacro{\prefCO}{1.0} \pgfmathsetmacro{\nfacCO}{3}
  \pgfmathsetmacro{\prefCS}{1.0} \pgfmathsetmacro{\nfacCS}{8}

  % chi_CS products: CH=0, CC=1, CO=0, CS=0.5
  \pgfmathsetmacro{\chiCSCH}{0.0}
  \pgfmathsetmacro{\chiCSCC}{1.0}
  \pgfmathsetmacro{\chiCSCO}{0.0}
  \pgfmathsetmacro{\chiCSCS}{0.5}

  \begin{axis}[
    width=0.95\linewidth,
    height=0.60\linewidth,
    xmin=0.65, xmax=2.45,
    ymin=-0.010, ymax=0.0003,
    xlabel={$r$ (\AA)},
    ylabel={Total correction $\Delta r(r)$ (\AA)},
    grid=both,
    legend pos=south east,
    legend cell align=left,
    tick label style={/pgf/number format/fixed},
    every axis plot/.append style={very thick}
  ]

    % helper: fcoord(r; rcov) = 0.5*(1 - erf((r-(rcov+0.35))/sig))
    % A(r) = -K*rcov*pref*nfac * fcoord
    % gauss(r; rcov) = exp(-((r - rcov + dr)/sig)^2)
    % total = A(r) * (1 - chiCS * gauss)

    % CH
    \addplot[domain=\rcovCH-0.40:\rcovCH+0.60, samples=320, orange]
    { ( -\K*\rcovCH*\prefCH*\nfacCH * 0.5*(1 - erf((x - (\rcovCH+0.35))/\sig)) )
      * ( 1 - \chiCSCH * exp(-((x - \rcovCH + \dr)/\sig)^2) ) };
    \addlegendentry{C--H}

    % CC
    \addplot[domain=\rcovCC-0.40:\rcovCC+0.60, samples=320, blue]
    { ( -\K*\rcovCC*\prefCC*\nfacCC * 0.5*(1 - erf((x - (\rcovCC+0.35))/\sig)) )
      * ( 1 - \chiCSCC * exp(-((x - \rcovCC + \dr)/\sig)^2) ) };
    \addlegendentry{C--C}

    % CO
    \addplot[domain=\rcovCO-0.40:\rcovCO+0.60, samples=320, teal!60!black]
    { ( -\K*\rcovCO*\prefCO*\nfacCO * 0.5*(1 - erf((x - (\rcovCO+0.35))/\sig)) )
      * ( 1 - \chiCSCO * exp(-((x - \rcovCO + \dr)/\sig)^2) ) };
    \addlegendentry{C--O}

    % CS
    \addplot[domain=\rcovCS-0.40:\rcovCS+0.60, samples=320, purple]
    { ( -\K*\rcovCS*\prefCS*\nfacCS * 0.5*(1 - erf((x - (\rcovCS+0.35))/\sig)) )
      * ( 1 - \chiCSCS * exp(-((x - \rcovCS + \dr)/\sig)^2) ) };
    \addlegendentry{C--S}

    % Vertical markers at r = rcov + 0.35 (beyond which fcoord -> 0)
    \addplot[gray, densely dashed] coordinates {(\rcovCH+0.35,-0.010) (\rcovCH+0.35,0.0003)};
    \addplot[gray, densely dashed] coordinates {(\rcovCC+0.35,-0.010) (\rcovCC+0.35,0.0003)};
    \addplot[gray, densely dashed] coordinates {(\rcovCO+0.35,-0.010) (\rcovCO+0.35,0.0003)};
    \addplot[gray, densely dashed] coordinates {(\rcovCS+0.35,-0.010) (\rcovCS+0.35,0.0003)};

  \end{axis}
\end{tikzpicture}
\caption{BDPCS3 total corrections $\Delta r(r)$ with $K=0.006$, $\Delta r=0.17$~\AA, $\sigma=0.057$~\AA. For each bond the correction smoothly goes to zero for $r \gtrsim r^{\mathrm{cov}}+0.35$~\AA\ (vertical dashed lines) because $f_{\mathrm{coord}}(r)\to 0$ as $\erf\to 1$.}
\label{fig:bdpcs3-centered}
